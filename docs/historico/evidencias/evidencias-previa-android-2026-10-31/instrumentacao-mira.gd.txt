extends Node2D

@onready var nave = get_parent().get_parent()
var pontos := PackedVector2Array()
var _qa_amostras: Array[Dictionary] = []
var _qa_aquecimento := 0
var _qa_lote := 0
var _qa_ultimo_tick := 0


func _ready() -> void:
	nave.get_node("Entrada").mira_encerrada.connect(limpar)
	nave.get_node("Entrada").impulso_solicitado.connect(func(_impulso: Vector2): limpar())


func limpar() -> void:
	_qa_amostras.clear()
	_qa_aquecimento = 0
	_qa_ultimo_tick = 0
	pontos.clear()
	queue_redraw()


func _physics_process(_delta: float) -> void:
	if not nave.entrada.pode_controlar() or nave.entrada.gesto.contato_ativo == -1:
		limpar()
		return
	var qa_inicio: int = Time.get_ticks_usec()
	pontos = nave.prever_trajetoria()
	var qa_custo: int = Time.get_ticks_usec() - qa_inicio
	_qa_registrar(qa_custo)
	queue_redraw()


func _qa_registrar(custo: int) -> void:
	if pontos.size() < 2:
		return
	var agora: int = Time.get_ticks_usec()
	_qa_aquecimento += 1
	if _qa_aquecimento > 60:
		_qa_amostras.append({"custo_us": custo, "intervalo_us": agora - _qa_ultimo_tick,
			"fps_engine": Engine.get_frames_per_second(), "pontos": pontos.size(),
			"condicao": int(nave.condicao), "combustivel": int(nave.combustivel),
			"tempo": float(nave.get_parent().partida.tempo_restante)})
	_qa_ultimo_tick = agora
	if _qa_amostras.size() == 300:
		var arquivo := FileAccess.open("user://previa-qa-%d.json" % _qa_lote, FileAccess.WRITE)
		if arquivo == null:
			push_error("QA: não foi possível gravar medição da prévia")
			return
		arquivo.store_string(JSON.stringify({"schema_version": 1, "aquecimento": 60,
			"fase": str(nave.get_parent().nivel_res.id), "amostras": _qa_amostras}, "", true, true))
		arquivo.close()
		_qa_lote += 1
		_qa_amostras.clear()


func _draw() -> void:
	var entrada = nave.get_node("Entrada")
	if not entrada.pode_controlar() or entrada.gesto.contato_ativo == -1:
		return
	var origem: Vector2 = entrada.gesto.origem
	var valido: bool = not entrada.gesto.obter_impulso().is_zero_approx()
	var cor := Color("82f0d1") if valido else Color("f5cc77")
	draw_arc(origem, entrada.zona_morta, 0, TAU, 40, Color("0b1423"), 3)
	draw_arc(origem, entrada.zona_morta, 0, TAU, 40, cor, 1)
	if not valido:
		draw_line(origem - Vector2(3, 3), origem + Vector2(3, 3), cor, 1)
		draw_line(origem - Vector2(-3, 3), origem + Vector2(-3, 3), cor, 1)
	var transformacao: Transform2D = nave.get_canvas_transform()
	for indice in range(1, pontos.size()):
		draw_line(transformacao * pontos[indice - 1], transformacao * pontos[indice], Color("0b1423"), 3)
		draw_line(transformacao * pontos[indice - 1], transformacao * pontos[indice], Color("b8f5e2"), 1)
	if pontos.size() > 1:
		draw_circle(transformacao * pontos[-1], 2, cor, false, 1)
